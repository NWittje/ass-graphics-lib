cmake_minimum_required(VERSION 3.25)
project(ass_graphics_lib VERSION 1.4.5 LANGUAGES C CXX RC)
include(cmake/VersionInfo.cmake)
set(CMAKE_SUPPRESS_REGENERATION ON)
add_link_options(/INCREMENTAL:NO)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ass_graphics_lib requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Every native binary owns its CRT allocations. Only handles, POD and borrowed
# buffers cross the ABI; the application never frees DLL-owned storage.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(ASS_GRAPHICS_BUILD_EXAMPLES "Build example applications" ON)
# Verification tooling is local-only and is intentionally absent from the repository.
set(ASS_GRAPHICS_LOCAL_TESTS OFF)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/local-targets.cmake")
  set(ASS_GRAPHICS_LOCAL_TESTS ON)
endif()
option(ASS_GRAPHICS_BUILD_TESTS "Build locally available verification" ${ASS_GRAPHICS_LOCAL_TESTS})
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
find_program(FXC fxc HINTS "C:/Program Files (x86)/Windows Kits/10/bin/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/x64" REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
foreach(SPEC "scene_vs;SceneVS;vs_5_0" "scene_ps;ScenePS;ps_5_0" "quad_vs;QuadVS;vs_5_0" "blur_ps;BlurPS;ps_5_0")
  list(GET SPEC 0 NAME)
  list(GET SPEC 1 ENTRY)
  list(GET SPEC 2 PROFILE)
  set(HEADER "${CMAKE_BINARY_DIR}/generated/${NAME}.h")
  add_custom_command(OUTPUT "${HEADER}" COMMAND "${FXC}" /nologo /O3 /T ${PROFILE} /E ${ENTRY} /Vn ${NAME} /Fh "${HEADER}" "${CMAKE_CURRENT_SOURCE_DIR}/resources/scene.hlsl" DEPENDS resources/scene.hlsl VERBATIM)
  list(APPEND SHADER_HEADERS "${HEADER}")
endforeach()
configure_file(resources/library.rc.in "${CMAKE_BINARY_DIR}/generated/library.rc" @ONLY)
foreach(SPEC "display_vs;DisplayVS;vs_5_0" "display_ps;DisplayPS;ps_5_0")
 list(GET SPEC 0 NAME)
 list(GET SPEC 1 ENTRY)
 list(GET SPEC 2 PROFILE)
 set(HEADER "${CMAKE_BINARY_DIR}/generated/${NAME}.h")
 add_custom_command(OUTPUT "${HEADER}" COMMAND "${FXC}" /nologo /O3 /T ${PROFILE} /E ${ENTRY} /Vn ${NAME} /Fh "${HEADER}" "${CMAKE_CURRENT_SOURCE_DIR}/resources/display.hlsl" DEPENDS resources/display.hlsl VERBATIM)
 list(APPEND SHADER_HEADERS "${HEADER}")
endforeach()
add_library(ass_graphics SHARED src/core.cpp src/font_dependency.cpp src/application.cpp src/refresh.cpp src/settings.cpp src/graphics2d.cpp src/charts.cpp src/graphics3d.cpp src/display.cpp src/allocations.cpp ${SHADER_HEADERS} "${CMAKE_BINARY_DIR}/generated/library.rc")
add_library(ass::graphics ALIAS ass_graphics)
set_target_properties(ass_graphics PROPERTIES OUTPUT_NAME ass_graphics_lib EXPORT_NAME graphics ASS_VERSION_STAMPED TRUE)
target_include_directories(ass_graphics PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(ass_graphics PRIVATE ASS_GRAPHICS_BUILD UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(ass_graphics PRIVATE /W4 /permissive- /EHsc /utf-8 /Zi)
target_link_options(ass_graphics PRIVATE /DEBUG:FULL)
target_link_libraries(ass_graphics PRIVATE d2d1 dwrite d3d11 dxgi windowscodecs ole32 comctl32 shell32 user32 gdi32)
if(ASS_GRAPHICS_BUILD_EXAMPLES)
  foreach(NAME basic tray gallery)
    add_executable(ass_${NAME} WIN32 examples/${NAME}.cpp examples/app.manifest)
    target_link_libraries(ass_${NAME} PRIVATE ass::graphics)
    target_compile_definitions(ass_${NAME} PRIVATE UNICODE _UNICODE NOMINMAX)
    # Ship matching example symbols in both Debug and optimized Release.
    target_compile_options(ass_${NAME} PRIVATE /utf-8 /Zi)
    target_link_options(ass_${NAME} PRIVATE /DEBUG:FULL)
  endforeach()
endif()
if(ASS_GRAPHICS_BUILD_TESTS)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/local-targets.cmake")
    message(FATAL_ERROR "Verification sources are not part of this source-only repository")
  endif()
  include(tests/local-targets.cmake)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS ass_graphics EXPORT AssGraphicsTargets RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY licenses/ DESTINATION licenses)
install(FILES managed/Ass.Graphics.Source.props DESTINATION bindings)
foreach(BINDING Ass.Graphics Ass.Graphics.WinForms)
  install(DIRECTORY managed/${BINDING}/ DESTINATION bindings/${BINDING}
          FILES_MATCHING PATTERN "*.cs" PATTERN "bin" EXCLUDE PATTERN "obj" EXCLUDE)
endforeach()
install(FILES "$<TARGET_PDB_FILE:ass_graphics>" DESTINATION symbols)
install(EXPORT AssGraphicsTargets NAMESPACE ass:: DESTINATION lib/cmake/AssGraphics)
configure_package_config_file(cmake/AssGraphicsConfig.cmake.in "${CMAKE_BINARY_DIR}/AssGraphicsConfig.cmake" INSTALL_DESTINATION lib/cmake/AssGraphics)
write_basic_package_version_file("${CMAKE_BINARY_DIR}/AssGraphicsConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_BINARY_DIR}/AssGraphicsConfig.cmake" "${CMAKE_BINARY_DIR}/AssGraphicsConfigVersion.cmake" DESTINATION lib/cmake/AssGraphics)
ass_stamp_directory("${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_VERSION}")
